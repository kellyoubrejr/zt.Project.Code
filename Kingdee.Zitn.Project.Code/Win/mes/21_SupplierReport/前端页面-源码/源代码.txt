<script>
(function() {
    const VERSION = 'v4.6.4';
    const CAT_MAP = new Map([
        ['LS01','临时供应商'],['WX13','基片供应商'],['WX14','管壳供应商'],['WX15','传感器供应商'],
        ['WX16','机加工供应商'],['WX17','PCB供应商'],['WX19','加表供应商'],['WX20','芯片供应商'],
        ['ZC01','集成电路供应商'],['ZC02','芯片供应商'],['ZC03','敏感器件供应商'],['ZC06','元器件供应商'],['ZC07','连接器供应商']
    ]);
    const IGNORE_CATEGORIES = new Set(['加表供应商']);
    function getCat(code) { const p = String(code||'').trim().substring(0,4); return CAT_MAP.get(p)||'其他/未分类'; }
    function isQual(s) { return ['合格','pass','ok','合格品'].includes(String(s||'').trim().toLowerCase()); }

    const FIELD_ALIASES = {
        supplierCode: ['供应商编码', '供应商代码', 'supplier_code'],
        name: ['供应商名称', 'supplier_name', '名称'],
        inspectQty: ['检验数量', '总数量', '数量', '来料数量', 'inspect_qty'],
        batchNo: ['批号', '批次号', 'batch_no'],
        startDate: ['检验开始日期', '开始日期'],
        endDate: ['检验结束日期', '结束日期', '检验完成日期'],
        defectQty: ['不合格数', '不良数', '不合格数量', 'defect_qty'],
        status: ['状态', '判定', 'status'],
        decision: ['使用决策备注', '决策备注', '备注']
    };
    function getField(row, aliases) {
        for (let key of aliases) if (row[key] !== undefined && row[key] !== '') return row[key];
        return '';
    }

    function parseMonthFromFilename(filename) {
        const m1 = filename.match(/26\.(\d{1,2})月/);
        if (m1) return `2026-${String(parseInt(m1[1])).padStart(2,'0')}`;
        const m2 = filename.match(/2026[年\-](\d{1,2})/);
        if (m2) return `2026-${String(parseInt(m2[1])).padStart(2,'0')}`;
        return null;
    }

    function merge618(rows) {
        const normal = [], special = [];
        for (let r of rows) {
            const code = String(getField(r, FIELD_ALIASES.supplierCode)).trim();
            (code === '618所' || code === '618' ? special : normal).push(r);
        }
        if (!special.length) return rows;
        const grp = new Map();
        for (let r of special) {
            const key = getField(r, ['批号','batch_no'])?.trim() || getField(r, FIELD_ALIASES.endDate)?.trim() || 'def_'+Math.random();
            if (!grp.has(key)) grp.set(key, {...r, 检验数量:0, 不合格数:0, rows:[]});
            const g = grp.get(key);
            g.检验数量 = (Number(g.检验数量)||0) + (Number(r.检验数量)||0);
            g.不合格数 = (Number(g.不合格数)||0) + (Number(r.不合格数)||0);
            g.rows.push(r);
            g.状态 = g.不合格数 > 0 ? '不合格' : '合格';
        }
        return [...normal, ...Array.from(grp.values()).map(g => { const {rows,...rest}=g; return rest; })];
    }

    function computeStats(rows) {
        const map = new Map();
        for (let r of rows) {
            const code = String(getField(r, FIELD_ALIASES.supplierCode)).trim();
            if (!code) continue;
            const qty = Math.floor(Number(getField(r, FIELD_ALIASES.inspectQty)) || 0);
            const defect = Math.floor(Number(getField(r, FIELD_ALIASES.defectQty)) || 0);
            const statusVal = getField(r, FIELD_ALIASES.status);
            const qual = isQual(statusVal);
            const name = getField(r, FIELD_ALIASES.name) || code;
            if (!map.has(code)) map.set(code, {code,name,totalQty:0,qualQty:0,batch:0,qualBatch:0,cat:getCat(code)});
            const s = map.get(code);
            s.totalQty += qty;
            s.qualQty += (qty - defect);
            s.batch++;
            if (qual) s.qualBatch++;
            if (name !== code && s.name === code) s.name = name;
        }
        const suppliers = Array.from(map.values()).map(s => ({
            ...s,
            batchQualRate: s.batch ? s.qualBatch/s.batch : 0,
            qtyQualRate: s.totalQty ? s.qualQty/s.totalQty : 0,
        }));
        const tb = suppliers.reduce((a,s)=>a+s.batch,0), tq = suppliers.reduce((a,s)=>a+s.totalQty,0);
        const tqb = suppliers.reduce((a,s)=>a+s.qualBatch,0), tqq = suppliers.reduce((a,s)=>a+s.qualQty,0);
        return { suppliers, totalBatch:tb, totalQty:tq, totalQualBatch:tqb, totalQualQty:tqq,
            overallBatchQualRate: tb?tqb/tb:0, overallQtyQualRate: tq?tqq/tq:0 };
    }

    function catStats(suppliers) {
        const m = new Map();
        for (let s of suppliers) {
            const cat = s.cat;
            if (IGNORE_CATEGORIES.has(cat)) continue;
            if (!m.has(cat)) m.set(cat, {cat, tb:0, tq:0, qb:0, qq:0});
            const o = m.get(cat);
            o.tb += s.batch; o.tq += s.totalQty; o.qb += s.qualBatch; o.qq += s.qualQty;
        }
        return Array.from(m.values()).map(o => ({
            ...o,
            batchQualRate: o.tb?o.qb/o.tb:0,
            qtyQualRate: o.tq?o.qq/o.tq:0,
        }));
    }

    let raw2025=[], raw2026ByMonth=new Map(), latestMonth=null, reportData=null;

    async function loadFile(file) {
        const data = await new Promise((res,rej)=>{
            const r = new FileReader();
            r.onload = e => { const wb = XLSX.read(new Uint8Array(e.target.result),{type:'array'}); res(XLSX.utils.sheet_to_json(wb.Sheets[wb.SheetNames[0]],{defval:''})); };
            r.onerror = rej; r.readAsArrayBuffer(file);
        });
        if (!data.length) throw new Error('文件无数据');
        return data;
    }

    document.getElementById('file2025').addEventListener('change', async function(){
        const f=this.files[0]; if(!f){raw2025=[]; return;}
        try{ raw2025 = merge618(await loadFile(f)); document.getElementById('status2025').innerHTML='✅ 2025数据已加载'; }
        catch(e){ raw2025=[]; document.getElementById('status2025').innerHTML='❌ '+e.message; }
    });
    document.getElementById('files2026').addEventListener('change', async function(){
        raw2026ByMonth.clear(); const files=Array.from(this.files); let total=0, errors=[];
        for (let file of files) {
            try {
                const data = merge618(await loadFile(file));
                const month = parseMonthFromFilename(file.name);
                if (!month) { errors.push(`${file.name}: 无法识别月份`); continue; }
                if (!raw2026ByMonth.has(month)) raw2026ByMonth.set(month, []);
                raw2026ByMonth.get(month).push(...data);
                total += data.length;
            } catch(e) { errors.push(`${file.name}: ${e.message}`); }
        }
        const months = [...raw2026ByMonth.keys()].sort();
        latestMonth = months.length ? months[months.length-1] : null;
        let msg = months.length ? `✅ 已加载 ${months.length} 个月份，共${total}条。` : '';
        if (errors.length) msg += '<br>⚠️ ' + errors.join('<br>');
        document.getElementById('status2026').innerHTML = msg || '未加载数据';
    });

    function compute2025Avg() {
        const groups = new Map();
        for (let r of raw2025) {
            let dateStr = getField(r, FIELD_ALIASES.endDate);
            if (!dateStr) dateStr = getField(r, ['单据日期', '检验开始日期']);
            if (!dateStr) continue;
            const ymMatch = String(dateStr).match(/(\d{4})[-/年](\d{1,2})/);
            if (!ymMatch || ymMatch[1] !== '2025') continue;
            const ym = `${ymMatch[1]}-${ymMatch[2].padStart(2,'0')}`;
            if (!groups.has(ym)) groups.set(ym, []);
            groups.get(ym).push(r);
        }
        if (groups.size === 0) return null;
        const monthStats = [];
        for (let [ym, rows] of groups) monthStats.push({month:ym, ...computeStats(rows), catStats:catStats(computeStats(rows).suppliers)});
        const n = monthStats.length;
        const avgB = monthStats.reduce((a,m)=>a+m.overallBatchQualRate,0)/n;
        const avgQ = monthStats.reduce((a,m)=>a+m.overallQtyQualRate,0)/n;
        const catMap = new Map();
        for (let m of monthStats) for (let c of m.catStats) {
            if (IGNORE_CATEGORIES.has(c.cat)) continue;
            if (!catMap.has(c.cat)) catMap.set(c.cat, []);
            catMap.get(c.cat).push({b:c.batchQualRate, q:c.qtyQualRate});
        }
        const catAvg = [];
        for (let [cat, rates] of catMap) catAvg.push({
            cat,
            avgB: rates.reduce((a,r)=>a+r.b,0)/rates.length,
            avgQ: rates.reduce((a,r)=>a+r.q,0)/rates.length,
        });
        return { monthStats, avgB, avgQ, catAvg };
    }

    function runAnalysis() {
        const a2025 = raw2025.length ? compute2025Avg() : null;
        const m2026 = [...raw2026ByMonth.keys()].sort();
        const monthly2026 = m2026.map(m => {
            const stats = computeStats(raw2026ByMonth.get(m));
            return { month:m, ...stats, catStats:catStats(stats.suppliers) };
        });
        const latest = m2026[m2026.length-1] || null;
        const latestStats = latest ? monthly2026.find(x=>x.month===latest) : null;
        const all2026Rows = [];
        for (let rows of raw2026ByMonth.values()) all2026Rows.push(...rows);
        const overall2026 = all2026Rows.length ? computeStats(all2026Rows) : null;
        const overall2026Cat = overall2026 ? catStats(overall2026.suppliers).filter(c => !IGNORE_CATEGORIES.has(c.cat) && c.tb>0) : [];
        const allCats = new Set();
        if (a2025) a2025.catAvg.forEach(c => { if (!IGNORE_CATEGORIES.has(c.cat)) allCats.add(c.cat); });
        monthly2026.forEach(m => m.catStats.forEach(c => { if (!IGNORE_CATEGORIES.has(c.cat) && c.tb>0) allCats.add(c.cat); }));
        overall2026Cat.forEach(c => allCats.add(c.cat));
        const sortedCats = [...allCats].sort();
        const catTrend = {};
        for (let cat of sortedCats) {
            catTrend[cat] = { avg2025B:null, avg2025Q:null, months:{} };
            if (a2025) {
                const ca = a2025.catAvg.find(c=>c.cat===cat);
                if (ca) { catTrend[cat].avg2025B = ca.avgB; catTrend[cat].avg2025Q = ca.avgQ; }
            }
            for (let m of monthly2026) {
                const cs = m.catStats.find(c=>c.cat===cat);
                if (cs) catTrend[cat].months[m.month] = { b:cs.batchQualRate, q:cs.qtyQualRate };
            }
        }
        return { a2025, monthly2026, m2026, latest, latestStats, overall2026, overall2026Cat, sortedCats, catTrend };
    }

    function renderReport(d) {
        reportData = d;
        document.getElementById('reportContainer').style.display='block';
        const { a2025, monthly2026, m2026, latest, latestStats, overall2026Cat, sortedCats, catTrend } = d;
        document.getElementById('reportTitleLabel').textContent = `${latest?.replace('2026-','2026年')}月 趋势报告`;

        let cards = '';
        if (latestStats) {
            cards += `<div class="stat-item"><div class="stat-value">${latestStats.totalBatch}</div><div class="stat-label">${latest}总批次</div></div>`;
            cards += `<div class="stat-item"><div class="stat-value">${latestStats.totalQty.toLocaleString()}</div><div class="stat-label">总数量</div></div>`;
            cards += `<div class="stat-item"><div class="stat-value">${(latestStats.overallBatchQualRate*100).toFixed(1)}%</div><div class="stat-label">批次合格率</div></div>`;
            cards += `<div class="stat-item"><div class="stat-value">${(latestStats.overallQtyQualRate*100).toFixed(1)}%</div><div class="stat-label">数量合格率</div></div>`;
        }
        if (a2025) {
            cards += `<div class="stat-item" style="background:#fef9e6;"><div class="stat-value">${(a2025.avgB*100).toFixed(1)}%</div><div class="stat-label">2025月均批次合格率</div></div>`;
            cards += `<div class="stat-item" style="background:#fef9e6;"><div class="stat-value">${(a2025.avgQ*100).toFixed(1)}%</div><div class="stat-label">2025月均数量合格率</div></div>`;
        }
        document.getElementById('summaryStats').innerHTML = cards;
        if (latestStats && a2025) {
            const db = (latestStats.overallBatchQualRate - a2025.avgB)*100;
            const dq = (latestStats.overallQtyQualRate - a2025.avgQ)*100;
            document.getElementById('latestHighlight').innerHTML = `📌 ${latest} vs 2025月平均：批次合格率 <b style="color:${db>=0?'#166534':'#991b1b'}">${db>=0?'↑':'↓'}${Math.abs(db).toFixed(1)}个百分点</b>，数量合格率 <b style="color:${dq>=0?'#166534':'#991b1b'}">${dq>=0?'↑':'↓'}${Math.abs(dq).toFixed(1)}个百分点</b>`;
        } else {
            document.getElementById('latestHighlight').innerHTML = '';
        }

        renderOverall(a2025, monthly2026, m2026);
        renderCategoryTrend(a2025, monthly2026, m2026, sortedCats, catTrend);
        renderCatLatest(latestStats);
        renderCatFull(a2025, monthly2026, m2026, overall2026Cat, sortedCats, catTrend);
        renderMachiningTrend(a2025, monthly2026, m2026);
    }

    function renderOverall(a2025, monthly, months) {
        const dom = document.getElementById('overallTrendChart');
        if (dom) { const inst = echarts.getInstanceByDom(dom); if (inst) inst.dispose(); }
        const chart = echarts.init(dom);
        const monthLabels = months.map(m => m.replace('2026-','')+'月');
        const batchData = months.map(m => { const s = monthly.find(x=>x.month===m); return s?+(s.overallBatchQualRate*100).toFixed(1):null; });
        const qtyData = months.map(m => { const s = monthly.find(x=>x.month===m); return s?+(s.overallQtyQualRate*100).toFixed(1):null; });
        if (a2025) {
            monthLabels.push('2025月平均');
            batchData.push(+(a2025.avgB*100).toFixed(1));
            qtyData.push(+(a2025.avgQ*100).toFixed(1));
        }
        chart.setOption({
            tooltip: { trigger: 'axis' },
            legend: { data: ['批次合格率(%)', '数量合格率(%)'] },
            xAxis: { data: monthLabels, axisLabel: { rotate: 30 } },
            yAxis: { name: '合格率(%)', max: 100 },
            series: [
                { name: '批次合格率(%)', type: 'bar', data: batchData, color: '#3498db', label: { show: true, position: 'top' } },
                { name: '数量合格率(%)', type: 'line', data: qtyData, color: '#2ecc71', symbol: 'circle', lineStyle: { width: 2 }, label: { show: true, position: 'top' } }
            ],
            grid: { containLabel: true }
        });

    function renderCatFull(a2025, monthly, months, overallCat, cats, trend) {

            const t = trend[cat];
            const oc = overallCat.find(c=>c.cat===cat);
            body += '<tr><td>'+cat+'</td>';
            body += `<td>${t.avg2025B!==null?(t.avg2025B*100).toFixed(1)+'%':'—'}</td><td>${t.avg2025Q!==null?(t.avg2025Q*100).toFixed(1)+'%':'—'}</td>`;

    }
